--------------------------------Finding the sub string count ---------------------------
s = "ABCDCDC"
sub = "CDC"

count = 0
for i in range(len(s) - len(sub) + 1):  # we are using +1 because index will start from 0 loop until substring fits
    if s[i:i+len(sub)] == sub:
        count += 1

print("Substring occurs:", count, "times")

basic
a = 'dskfdccd'
b = 'cadcbsc'

w = []

for i in a:
    if i not in b:
        w.append(i)

for j in b:
    if j not in a:
        w.append(j)

print(w)

b = 'cadcbsc'

w = list(set(a) ^ set(b))

^ (symmetric difference operator)
This operator finds elements that are in either set, but NOT in both
In simple words:        “Give me characters that appear in only one string.”

# binarysearch

mylist = [-1, 0, 3, 5, 9, 12]

def binarysearch(mylist, target):
    l = 0
    r = len(mylist) - 1

    while l <= r:
        m = (l + r) // 2

        if target == mylist[m]:
            return m
        elif target > mylist[m]:
            l = m + 1
        elif target < mylist[m]:
            r = m - 1

    return -1


r = binarysearch(mylist, -1)
print(r)

# Convert Second Letter of Every Word to Uppercase
# Input: "my name is dinesh"
# Output: "mY nAme iS dInesh"
# -----------------------------------

u_inp = "my name is dinesh"

words = u_inp.split()

result = []

for word in words:
    if len(word) > 1:
        result.append(word[0] + word[1].upper() + word[2:])
    else:
        result.append(word)

print(" ".join(result))


# Using List Comprehension

u_inp = "my name is dinesh im from dharmapuri"

result = [
    word[0] + word[1].upper() + word[2:] if len(word) > 1 else word
    for word in u_inp.split()
]

print(" ".join(result))
